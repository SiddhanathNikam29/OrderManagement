-- Products Table
CREATE TABLE Products (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Name NVARCHAR(200) NOT NULL,
    Description NVARCHAR(500) NULL,
    UnitPrice DECIMAL(18,2) NOT NULL CONSTRAINT CHK_Product_UnitPrice_NonNegative CHECK (UnitPrice >= 0),
    IsTaxable BIT NOT NULL DEFAULT 1,
    Category NVARCHAR(100) NOT NULL,
    CreatedAt DATETIME2 NOT NULL DEFAULT GETUTCDATE(),
    IsActive BIT NOT NULL DEFAULT 1
);
GO

-- Orders Table
CREATE TABLE Orders (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    OrderNumber NVARCHAR(50) NOT NULL UNIQUE,
    CustomerName NVARCHAR(200) NOT NULL,
    CustomerEmail NVARCHAR(200) NULL,
    OrderDate DATETIME2 NOT NULL DEFAULT GETUTCDATE(),
    Subtotal DECIMAL(18,2) NOT NULL DEFAULT 0,
    DiscountType NVARCHAR(20) NULL CONSTRAINT CHK_Order_DiscountType CHECK (DiscountType IN ('Percentage', 'Fixed', NULL)),
    DiscountValue DECIMAL(18,2) NULL CONSTRAINT CHK_Order_DiscountValue_NonNegative CHECK (DiscountValue >= 0 OR DiscountValue IS NULL),
    DiscountAmount DECIMAL(18,2) NOT NULL DEFAULT 0 CONSTRAINT CHK_Order_DiscountAmount_NonNegative CHECK (DiscountAmount >= 0),
    TaxableAmount DECIMAL(18,2) NOT NULL DEFAULT 0 CONSTRAINT CHK_Order_TaxableAmount_NonNegative CHECK (TaxableAmount >= 0),
    TaxAmount DECIMAL(18,2) NOT NULL DEFAULT 0 CONSTRAINT CHK_Order_TaxAmount_NonNegative CHECK (TaxAmount >= 0),
    Total DECIMAL(18,2) NOT NULL DEFAULT 0 CONSTRAINT CHK_Order_Total_NonNegative CHECK (Total >= 0),
    Status NVARCHAR(20) NOT NULL DEFAULT 'Active' CONSTRAINT CHK_Order_Status CHECK (Status IN ('Active', 'Completed', 'Cancelled', 'Deleted')),
    Version INT NOT NULL DEFAULT 1,
    CreatedAt DATETIME2 NOT NULL DEFAULT GETUTCDATE(),
    UpdatedAt DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);
GO

-- Order Items Table

-- ============================================================
-- vw_OrderSummary - Optimized summary view for read operations
-- ============================================================
CREATE VIEW vw_OrderSummary AS
SELECT 
    o.Id AS OrderId,
    o.OrderNumber,
    o.CustomerName,
    o.CustomerEmail,
    o.OrderDate,
    ISNULL(COUNT(oi.Id), 0) AS ItemCount,
    o.Subtotal,
    o.DiscountType,
    o.DiscountValue,
    o.DiscountAmount,
    o.TaxableAmount,
    o.TaxAmount,
    o.Total,
    o.Status,
    o.Version,
    o.UpdatedAt,
    CASE 
        WHEN o.DiscountType = 'Percentage' AND o.DiscountValue IS NOT NULL 
        THEN o.DiscountValue 
        WHEN o.DiscountType = 'Fixed' AND o.DiscountValue IS NOT NULL 
        THEN o.DiscountValue
        ELSE 0 
    END AS DiscountDisplayValue,
    CASE 
        WHEN o.DiscountType = 'Percentage' THEN '%'
        WHEN o.DiscountType = 'Fixed' THEN '$'
        ELSE ''
    END AS DiscountSymbol
FROM Orders o
LEFT JOIN OrderItems oi ON o.Id = oi.OrderId
WHERE o.Status != 'Deleted'
GROUP BY 
    o.Id, 
    o.OrderNumber, 
    o.CustomerName,
    o.CustomerEmail,
    o.OrderDate,
    o.Subtotal, 
    o.DiscountType, 
    o.DiscountValue, 
    o.DiscountAmount,
    o.TaxableAmount, 
    o.TaxAmount, 
    o.Total, 
    o.Status,
    o.Version,
    o.UpdatedAt;
GO

    IsTaxable,
    Category,
    CreatedAt,
    CASE WHEN IsTaxable = 1 THEN 'Taxable' ELSE 'Zero-Rated' END AS TaxStatus

CREATE VIEW vw_OrderDetails AS
SELECT 
    o.Id AS OrderId,
    o.OrderNumber,
    o.CustomerName,
    o.CustomerEmail,
    o.OrderDate,
    o.Subtotal,
    o.DiscountType,
    o.DiscountValue,
    o.DiscountAmount,
    o.TaxableAmount,
    o.TaxAmount,
    o.Total,
    o.Status,
    o.Version,
    o.UpdatedAt,
    oi.Id AS ItemId,
    oi.ProductId,
    oi.ProductName,
    oi.UnitPrice AS ItemUnitPrice,
    oi.Quantity,
    oi.LineTotal,
    p.IsTaxable AS ProductIsTaxable
FROM Orders o
INNER JOIN OrderItems oi ON o.Id = oi.OrderId
INNER JOIN Products p ON oi.ProductId = p.Id
WHERE o.Status != 'Deleted';
GO

-- ============================================================
-- sp_CreateOrder
-- Description: Creates a new order with generated order number
-- ============================================================
GO
CREATE PROCEDURE sp_CreateOrder
    @CustomerName NVARCHAR(200),
    @CustomerEmail NVARCHAR(200) = NULL,
    @OrderId INT OUTPUT,
    @OrderNumber NVARCHAR(50) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    
    BEGIN TRY
        BEGIN TRANSACTION;
        
        -- Generate Order Number
        SET @OrderNumber = 'ORD-' + 
                          FORMAT(GETUTCDATE(), 'yyyyMMddHHmmss') + 
                          '-' + 
                          RIGHT('0000' + CAST(ABS(CHECKSUM(NEWID())) % 10000 AS NVARCHAR(4)), 4);
        
        -- Insert Order
        INSERT INTO Orders (
            OrderNumber,
            CustomerName,
            CustomerEmail,
            OrderDate,
            Subtotal,
            DiscountAmount,
            TaxableAmount,
            TaxAmount,
            Total,
            Status,
            CreatedAt,
            UpdatedAt
        ) VALUES (
            @OrderNumber,
            @CustomerName,
            @CustomerEmail,
            GETUTCDATE(),
            0,
            0,
            0,
            0,
            0,
            'Active',
            GETUTCDATE(),
            GETUTCDATE()
        );
        
        SET @OrderId = SCOPE_IDENTITY();
        
        COMMIT TRANSACTION;
        
        -- Return result
        SELECT 
            @OrderId AS OrderId, 
            @OrderNumber AS OrderNumber;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        DECLARE @ErrorSeverity INT = ERROR_SEVERITY();
        DECLARE @ErrorState INT = ERROR_STATE();
        
        RAISERROR(@ErrorMessage, @ErrorSeverity, @ErrorState);
    END CATCH
END
GO

-- ============================================================
-- sp_RecalculateOrderTotals
-- Description: Recalculates all order totals (subtotal, tax, discount, total)
-- ============================================================
GO
CREATE PROCEDURE sp_RecalculateOrderTotals
    @OrderId INT
AS
BEGIN
    SET NOCOUNT ON;
    
    DECLARE @Subtotal DECIMAL(18,2);
    DECLARE @TaxableAmount DECIMAL(18,2);
    DECLARE @DiscountType NVARCHAR(20);
    DECLARE @DiscountValue DECIMAL(18,2);
    DECLARE @DiscountAmount DECIMAL(18,2);
    DECLARE @TaxAmount DECIMAL(18,2);
    DECLARE @Total DECIMAL(18,2);
    DECLARE @TAX_RATE DECIMAL(5,4) = 0.20; -- 20%
    
    -- Validate order exists
    IF NOT EXISTS (SELECT 1 FROM Orders WHERE Id = @OrderId)
    BEGIN
        RAISERROR('Order with ID %d does not exist', 16, 1, @OrderId);
        RETURN;
    END;
    
    -- Calculate Subtotal and Taxable Amount
    SELECT 
        @Subtotal = ISNULL(SUM(LineTotal), 0),
        @TaxableAmount = ISNULL(SUM(CASE WHEN p.IsTaxable = 1 THEN oi.LineTotal ELSE 0 END), 0)
    FROM OrderItems oi
    INNER JOIN Products p ON oi.ProductId = p.Id
    WHERE oi.OrderId = @OrderId;
    
    -- Get Discount
    SELECT 
        @DiscountType = DiscountType,
        @DiscountValue = DiscountValue
    FROM Orders
    WHERE Id = @OrderId;
    
    -- Calculate Discount Amount
    SET @DiscountAmount = 0;
    
    IF @DiscountType = 'Fixed' AND @DiscountValue IS NOT NULL AND @DiscountValue > 0
    BEGIN
        SET @DiscountAmount = CASE 
            WHEN @DiscountValue <= @Subtotal THEN @DiscountValue 
            ELSE @Subtotal 
        END;
    END
    ELSE IF @DiscountType = 'Percentage' AND @DiscountValue IS NOT NULL AND @DiscountValue > 0
    BEGIN
        SET @DiscountAmount = @Subtotal * (@DiscountValue / 100);
    END
    
    -- Calculate Tax (proportional to taxable items)
    IF @TaxableAmount > 0 AND @Subtotal > 0
    BEGIN
        DECLARE @DiscountProportion DECIMAL(18,6);
        SET @DiscountProportion = @TaxableAmount / @Subtotal;
        DECLARE @TaxableDiscount DECIMAL(18,2);
        SET @TaxableDiscount = @DiscountAmount * @DiscountProportion;
        DECLARE @DiscountedTaxableAmount DECIMAL(18,2);
        SET @DiscountedTaxableAmount = @TaxableAmount - @TaxableDiscount;
        SET @TaxAmount = @DiscountedTaxableAmount * @TAX_RATE;
    END
    ELSE
    BEGIN
        SET @TaxAmount = 0;
    END
    
    -- Calculate Total
    SET @Total = (@Subtotal - @DiscountAmount) + @TaxAmount;
    
    -- Update Order
    UPDATE Orders
    SET 
        Subtotal = ROUND(@Subtotal, 2),
        DiscountAmount = ROUND(@DiscountAmount, 2),
        TaxableAmount = ROUND(@TaxableAmount, 2),
        TaxAmount = ROUND(@TaxAmount, 2),
        Total = ROUND(@Total, 2),
        UpdatedAt = GETUTCDATE(),
        Version = Version + 1
    WHERE Id = @OrderId;
END
GO

-- ============================================================
-- sp_AddItemToOrder
-- Description: Adds or updates item in order and recalculates totals
-- ============================================================
GO
CREATE OR ALTER PROCEDURE sp_AddItemToOrder
    @OrderId INT,
    @ProductId INT,
    @Quantity INT
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        DECLARE @UnitPrice DECIMAL(18,2);
        DECLARE @ProductName NVARCHAR(200);
        
        BEGIN TRANSACTION;
        
        -- Validate Order
        IF NOT EXISTS (SELECT 1 FROM Orders WHERE Id = @OrderId AND Status = 'Active')
        BEGIN
            RAISERROR('Order not found or not active', 16, 1);
            RETURN;
        END;
        
        -- Validate Quantity
        IF @Quantity <= 0
        BEGIN
            RAISERROR('Quantity must be greater than 0', 16, 1);
            RETURN;
        END;
        
        -- Get Product details
        SELECT 
            @UnitPrice = UnitPrice,
            @ProductName = Name
        FROM Products
        WHERE Id = @ProductId AND IsActive = 1;
        
        IF @UnitPrice IS NULL
        BEGIN
            RAISERROR('Product not found or inactive', 16, 1);
            RETURN;
        END;
        
        -- ✅ Insert WITHOUT IsTaxable
        INSERT INTO OrderItems (
            OrderId,
            ProductId,
            ProductName,
            UnitPrice,
            Quantity,
            LineTotal,
            CreatedAt
        ) VALUES (
            @OrderId,
            @ProductId,
            @ProductName,
            @UnitPrice,
            @Quantity,
            @UnitPrice * @Quantity,
            GETUTCDATE()
        );
        
        -- Recalculate Order Totals
        EXEC sp_RecalculateOrderTotals @OrderId;
        
        COMMIT TRANSACTION;
        
        -- Return updated order details
        SELECT * FROM vw_OrderDetails WHERE OrderId = @OrderId ORDER BY ItemId;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        THROW;
    END CATCH
END
GO
-- ============================================================
-- sp_RemoveItemFromOrder
-- Description: Removes item from order and recalculates totals
-- ============================================================
GO
CREATE PROCEDURE sp_RemoveItemFromOrder
    @OrderId INT,
    @ItemId INT
AS
BEGIN
    SET NOCOUNT ON;
    
    BEGIN TRY
        BEGIN TRANSACTION;
        
        -- Validate order exists
        IF NOT EXISTS (SELECT 1 FROM Orders WHERE Id = @OrderId AND Status = 'Active')
        BEGIN
            RAISERROR('Order not found or not active', 16, 1);
            RETURN;
        END;
        
        -- Validate item exists in order
        IF NOT EXISTS (SELECT 1 FROM OrderItems WHERE Id = @ItemId AND OrderId = @OrderId)
        BEGIN
            RAISERROR('Item not found in this order', 16, 1);
            RETURN;
        END;
        
        -- Delete item
        DELETE FROM OrderItems
        WHERE Id = @ItemId AND OrderId = @OrderId;
        
        -- Recalculate totals
        EXEC sp_RecalculateOrderTotals @OrderId;
        
        COMMIT TRANSACTION;
        
        -- Return updated order details
        SELECT * FROM vw_OrderDetails WHERE OrderId = @OrderId ORDER BY ItemId;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        DECLARE @ErrorSeverity INT = ERROR_SEVERITY();
        DECLARE @ErrorState INT = ERROR_STATE();
        
        RAISERROR(@ErrorMessage, @ErrorSeverity, @ErrorState);
    END CATCH
END
GO

-- ============================================================
-- sp_ApplyDiscount
-- Description: Applies or removes discount from order and recalculates totals
-- ============================================================
GO
CREATE PROCEDURE sp_ApplyDiscount
    @OrderId INT,
    @DiscountType NVARCHAR(20), -- 'Percentage' or 'Fixed'
    @DiscountValue DECIMAL(18,2)
AS
BEGIN
    SET NOCOUNT ON;
    
    BEGIN TRY
        BEGIN TRANSACTION;
        
        -- Validate order exists
        IF NOT EXISTS (SELECT 1 FROM Orders WHERE Id = @OrderId AND Status = 'Active')
        BEGIN
            RAISERROR('Order not found or not active', 16, 1);
            RETURN;
        END;
        
        -- Validate discount parameters
        IF @DiscountType IS NOT NULL AND @DiscountType NOT IN ('Percentage', 'Fixed')
        BEGIN
            RAISERROR('Invalid discount type. Must be Percentage, Fixed, or NULL', 16, 1);
            RETURN;
        END;
        
        IF @DiscountValue < 0
        BEGIN
            RAISERROR('Discount value cannot be negative', 16, 1);
            RETURN;
        END;
        
        IF @DiscountType = 'Percentage' AND @DiscountValue > 100
        BEGIN
            RAISERROR('Percentage discount cannot exceed 100%', 16, 1);
            RETURN;
        END;
        
        -- Update discount
        IF @DiscountValue = 0 OR @DiscountType IS NULL
        BEGIN
            UPDATE Orders
            SET 
                DiscountType = NULL,
                DiscountValue = NULL,
                UpdatedAt = GETUTCDATE()
            WHERE Id = @OrderId;
        END
        ELSE
        BEGIN
            UPDATE Orders
            SET 
                DiscountType = @DiscountType,
                DiscountValue = @DiscountValue,
                UpdatedAt = GETUTCDATE()
            WHERE Id = @OrderId;
        END
        
        -- Recalculate totals
        EXEC sp_RecalculateOrderTotals @OrderId;
        
        COMMIT TRANSACTION;
        
        -- Return updated order details
        SELECT * FROM vw_OrderDetails WHERE OrderId = @OrderId ORDER BY ItemId;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        DECLARE @ErrorSeverity INT = ERROR_SEVERITY();
        DECLARE @ErrorState INT = ERROR_STATE();
        
        RAISERROR(@ErrorMessage, @ErrorSeverity, @ErrorState);
    END CATCH
END
GO

-- ============================================================
-- sp_GetAllOrders
-- Description: Returns paged list of all orders
-- ============================================================
GO
CREATE PROCEDURE sp_GetAllOrders
    @Page INT = 1,
    @PageSize INT = 10,
    @TotalCount INT OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    
    DECLARE @Offset INT = (@Page - 1) * @PageSize;
    
    -- Validate parameters
    IF @Page < 1 SET @Page = 1;
    IF @PageSize < 1 SET @PageSize = 10;
    IF @PageSize > 100 SET @PageSize = 100;
    
    -- Get total count
    SELECT @TotalCount = COUNT(*)
    FROM Orders
    WHERE Status != 'Deleted';
    
    -- Get paged results
    SELECT 
        OrderId,
        OrderNumber,
        CustomerName,
        CustomerEmail,
        OrderDate,
        ItemCount,
        Subtotal,
        DiscountType,
        DiscountValue,
        DiscountAmount,
        TaxableAmount,
        TaxAmount,
        Total,
        Status,
        Version,
        UpdatedAt,
        DiscountDisplayValue,
        DiscountSymbol
    FROM vw_OrderSummary
    WHERE Status != 'Deleted'
    ORDER BY OrderDate DESC
    OFFSET @Offset ROWS
    FETCH NEXT @PageSize ROWS ONLY;
END
GO

-- ============================================================
-- sp_SplitOrder
-- Description: Splits order total into equal shares with fair rounding
-- ============================================================
GO
CREATE PROCEDURE sp_SplitOrder
    @OrderId INT,
    @NumberOfShares INT,
    @SplitData NVARCHAR(MAX) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    
    DECLARE @Total DECIMAL(18,2);
    DECLARE @BaseAmount DECIMAL(18,2);
    DECLARE @RemainderCents INT;
    DECLARE @ShareIndex INT = 1;
    DECLARE @TotalSplit DECIMAL(18,2);
    DECLARE @Difference DECIMAL(18,2);
    
    -- Create temp table for shares
    CREATE TABLE #Shares (
        ShareNumber INT,
        Amount DECIMAL(18,2)
    );
    
    -- Get order total
    SELECT @Total = Total
    FROM Orders
    WHERE Id = @OrderId AND Status = 'Active';
    
    IF @Total IS NULL
    BEGIN
        RAISERROR('Order not found or not active', 16, 1);
        RETURN;
    END;
    
    IF @NumberOfShares <= 1
    BEGIN
        RAISERROR('Number of shares must be at least 2', 16, 1);
        RETURN;
    END;
    
    IF @NumberOfShares > 100
    BEGIN
        RAISERROR('Number of shares cannot exceed 100', 16, 1);
        RETURN;
    END;
    
    -- Calculate split
    SET @BaseAmount = FLOOR((@Total / @NumberOfShares) * 100) / 100;
    SET @RemainderCents = CAST(ROUND((@Total - (@BaseAmount * @NumberOfShares)) * 100, 0) AS INT);
    
    -- Generate shares
    WHILE @ShareIndex <= @NumberOfShares
    BEGIN
        DECLARE @Amount DECIMAL(18,2);
        SET @Amount = @BaseAmount;
        
        IF @ShareIndex <= @RemainderCents
        BEGIN
            SET @Amount = @Amount + 0.01;
        END;
        
        INSERT INTO #Shares (ShareNumber, Amount)
        VALUES (@ShareIndex, ROUND(@Amount, 2));
        
        SET @ShareIndex = @ShareIndex + 1;
    END;
    
    -- Verify and adjust last share if needed
    SELECT @TotalSplit = SUM(Amount) FROM #Shares;
    SET @Difference = ROUND(@Total - @TotalSplit, 2);
    
    IF @Difference != 0
    BEGIN
        UPDATE #Shares
        SET Amount = Amount + @Difference
        WHERE ShareNumber = @NumberOfShares;
    END;
    
    -- Build JSON result
    SET @SplitData = (
        SELECT 
            @Total AS TotalAmount,
            @NumberOfShares AS NumberOfShares,
            (
                SELECT 
                    ShareNumber,
                    ROUND(Amount, 2) AS Amount
                FROM #Shares
                ORDER BY ShareNumber
                FOR JSON PATH
            ) AS Shares
        FOR JSON PATH, WITHOUT_ARRAY_WRAPPER
    );
    
    -- Clean up
    DROP TABLE #Shares;
END
GO
